<script>
const $=id=>document.getElementById(id);
function enterApp(){ $('login').classList.remove('active'); $('main').classList.add('active'); }
function go(id){
 document.querySelectorAll('#main .content > .screen').forEach(s=>s.classList.remove('active'));
 $(id).classList.add('active');
 document.querySelectorAll('.nav').forEach(n=>n.classList.toggle('active',n.dataset.go===id));
 window.scrollTo({top:0,behavior:'smooth'});
}
function n(id){return Math.max(0,Number($(id).value)||0)}
function analyze(){
 let shower=n('shower'),wash=n('wash'),dish=n('dish'),garden=n('garden'),flush=n('flush'),car=n('car'),cook=n('cook'),meter=n('meter');
 let total=meter>0?meter:(shower*10+wash*70+dish*25+garden*12+flush*6+car*120+cook*12);
 let baseline=470, saved=Math.max(0,baseline-total), score=Math.max(50,Math.min(100,Math.round(100-(total-baseline)*.12)));
 let top=[['Shower',shower*10],['Laundry',wash*70],['Dishwashing',dish*25],['Gardening',garden*12],['Toilet',flush*6],['Car wash',car*120],['Cooking',cook*12]].sort((a,b)=>b[1]-a[1])[0];
 let predicted=Math.round(total*30);
 $('dashTotal').innerHTML=`${Math.round(total)} <span class="unit">L</span>`;
 $('dashSaved').textContent=Math.round(saved)+' L';
 $('dashScore').textContent=score+'/100';
 $('projected').textContent=predicted.toLocaleString()+' L';
 $('targetText').textContent=predicted.toLocaleString()+' / 15,000 L';
 $('targetBar').style.width=Math.min(100,predicted/15000*100)+'%';
 $('auditResult').classList.remove('hidden');
 $('auditResult').innerHTML=`<div class="row"><div><div class="label">Estimated daily consumption</div><div class="big" style="color:var(--teal);font-size:30px">${Math.round(total)} L</div></div><span class="badge">${score}/100</span></div>
 <div class="section"><h2>AI analysis</h2></div>
 <div class="alert ${total>550?'dangerbox':'success'}">${total>550?'⚠ Higher-than-usual consumption detected. Check your highest-use activity and unusual usage periods.':'✓ Great! Your estimated usage is within a conservation-friendly range.'}</div>
 <div class="row mt"><span class="sub">Highest-use activity</span><b>${top[0]} (${Math.round(top[1])} L)</b></div>
 <div class="row mt"><span class="sub">Projected monthly usage</span><b>${predicted.toLocaleString()} L</b></div>
 <button class="btn mt" onclick="go('insights')">View AI Recommendations →</button>`;
 $('coachTitle').textContent=top[0]==='Shower'?'Your shower is today’s biggest opportunity.':`${top[0]} is your biggest water-use opportunity.`;
 $('coachText').textContent=`Reducing this activity can improve your score and save water every month.`;
 toast('Water usage analyzed successfully ✦');
}
function rainCalc(){
 let roof=n('roof'),rain=n('rainfall'),eff=n('eff')/100,harvest=roof*rain*eff;
 let money=harvest*0.02;
 $('rainResult').innerHTML=`<div class="label" style="color:#d8ffff">Estimated annual harvest</div><div class="big">${Math.round(harvest).toLocaleString()} <span class="unit">L</span></div><div class="sub mt" style="color:#d8ffff">Potential annual water-value offset ≈ ₹${Math.round(money).toLocaleString()}</div>`;
}
function simulate(){let x=n('sim');$('simVal').textContent=x;$('simSave').textContent=(x*300)+' L'}
function toast(t){let e=$('toast');e.textContent=t;e.style.display='block';clearTimeout(window.tt);window.tt=setTimeout(()=>e.style.display='none',2200)}
function makeBars(id,vals,labels){let el=$(id),max=Math.max(...vals);el.innerHTML=vals.map((v,i)=>`<div class="barcol"><div class="barfill" style="height:${Math.max(12,v/max*115)}px"></div><small>${labels[i]}</small></div>`).join('')}
makeBars('homeChart',[390,420,450,410,470,440,420],['M','T','W','T','F','S','S']);
makeBars('analyticsChart',[390,420,450,410,470,440,420],['M','T','W','T','F','S','S']);
rainCalc();
